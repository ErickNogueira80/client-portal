💻 Queries & Consultas Úteis

Consulta para avaliar capacity PBI:

PowerBIDatasetsWorkspace
| where PowerBIWorkspaceId == "DB989EE3-0185-41CD-80A3-F1F56B065054"
| where OperationName in ("VertiPaqSEQueryBegin", "VertiPaqSEQueryEnd")
| extend ReportId = tostring(parse_json(ApplicationContext).Sources[0].ReportId)
| extend VisualId = tostring(parse_json(ApplicationContext).Sources[0].VisualId)
| summarize 
    StartTime = minif(TimeGenerated, OperationName == "VertiPaqSEQueryBegin"),
    EndTime = maxif(TimeGenerated, OperationName == "VertiPaqSEQueryEnd"),
    EventText = anyif(EventText, OperationName == "VertiPaqSEQueryBegin"),
    CpuTimeMs = maxif(CpuTimeMs, OperationName == "VertiPaqSEQueryEnd"),
    ExecutingUser = any(ExecutingUser),
    ArtifactName = any(ArtifactName),
    ReportId = any(ReportId),
    VisualId = any(VisualId)
    by XmlaRequestId
| where isnotempty(EndTime)
| extend DurationMs = datetime_diff('millisecond', EndTime, StartTime)
| summarize 
    TotalExecucoes = count(),
    AvgDurationMs = avg(DurationMs),
    MaxDurationMs = max(DurationMs),
    TotalCpuMs = sum(CpuTimeMs),
    Usuarios = dcount(ExecutingUser)
    by ArtifactName, ReportId, VisualId
| order by TotalCpuMs desc
